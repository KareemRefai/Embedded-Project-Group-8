void timer1_delay_us(unsigned int us) {
    unsigned int preload; // initial count value

    T1CON = 0x10;          //Timer1 OFF, internal clock, prescaler 1:2

    preload = 65536 - us; //count to 2^16 before overflow

    TMR1H = preload / 256; //high byte, Timer1 high
    TMR1L = preload % 256; //low byte, Timer1 Low

    PIR1 = PIR1 & 0xFE;    //clear bit 0 (TMR1IF)

    T1CON = T1CON | 0x01;  //set bit 0 (TMR1ON)

    while((PIR1 & 0x01) == 0);

    T1CON = T1CON & 0xFE;  //clear bit 0 (T1CON)
}

unsigned int servo_i;


// ================= ADC FOR LDR ON RA1 / AN1 =================
void ADC_Init_LDR_RA1() {
    ADCON1 = 0x80; //PORTA analog inputs (right justified)
    ADCON0 = 0x49; //RA1 selected
}

unsigned int read_ldr_RA1() {
    ADCON0 = 0x49;
    manual_delay(20); //charge capacitor

    ADCON0 = ADCON0 | 0x04; //start ATD conversion

    while(ADCON0 & 0x04);

    return ((unsigned int)ADRESH * 256) + ADRESL;
}


// ================= PWM INIT =================
// ENA = RC1 = CCP2
// ENB = RC2 = CCP1

void CCPPWM_init(void) {
    TRISC = TRISC & 0xF9;

    PR2 = 250; //set pwm period register

    CCP1CON = 0x0C;
    CCP2CON = 0x0C;

    CCPR1L = 0;
    CCPR2L = 0;

    T2CON = 0x07; //start Timer2, prescaler 1:16, enable pwm signals
}


// ================= MOTOR SPEED =================

void motor_R(unsigned char speed) {
    CCPR1L = speed;
}

void motor_L(unsigned char speed) {
    CCPR2L = speed;
}


// ================= MOTOR CONTROL =================
// IN1 = RD0
// IN2 = RD1
// IN3 = RD2
// IN4 = RD3

void stop_motors() {
    PORTD = PORTD & 0xF0; //clear RD0-RD3, turn off direction

    motor_L(0);
    motor_R(0);
}

void forward() { //RD0 = 1,RD2 = 1

    PORTD = PORTD | 0x05;
    PORTD = PORTD & 0xF5;

    motor_L(55);
    motor_R(55);
}

void turn_left() {

    PORTD = PORTD | 0x01;   //RD0 = 1
    PORTD = PORTD & 0xF1;   //RD1, RD2, RD3 = 0

    motor_L(60);
    motor_R(0);
}

void turn_right() {
    PORTD = PORTD | 0x04;   //RD2 = 1
    PORTD = PORTD & 0xF4;   //RD0, RD1, RD3 = 0

    motor_L(0);
    motor_R(60);
}
void turn_right1() { //RD0,RD2 = 1

     PORTD = PORTD | 0x05;
    PORTD = PORTD & 0xF5;

    motor_L(45);
    motor_R(65);
}
void hard_right() { //RD1,RD2 = 1
    PORTD = PORTD & 0xF0;
    PORTD = PORTD | 0x06;

    motor_L(70);
    motor_R(70);
}

void raise_servo_RD6() {

    for(servo_i = 0; servo_i < 100; servo_i++) {
        PORTD = PORTD | 0x40; //turn RD6 high
        timer1_delay_us(2000);
        PORTD = PORTD & ~0x40; //turn RD6 low
        timer1_delay_us((unsigned int)(20000 - 2000));
   }
}
